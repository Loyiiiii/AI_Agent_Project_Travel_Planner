Trip Intake Module - Class Diagram

class TripRequest {
  - requestId: string
  - origin: string
  - destination: string
  - startDate: string
  - endDate: string
  - numberOfTravelers: int
  - budget: double
  - currency: string
  - interests: vector<string>
  - travelPace: string
  - status: TripRequestStatus

  + getOrigin(): string
  + setOrigin(origin: string): void
  + getDestination(): string
  + setDestination(destination: string): void
  + getStartDate(): string
  + setStartDate(startDate: string): void
  + getEndDate(): string
  + setEndDate(endDate: string): void
  + getNumberOfTravelers(): int
  + setNumberOfTravelers(numberOfTravelers: int): void
  + getBudget(): double
  + setBudget(budget: double): void
  + getCurrency(): string
  + setCurrency(currency: string): void
  + getInterests(): vector<string>
  + setInterests(interests: vector<string>): void
  + getTravelPace(): string
  + setTravelPace(travelPace: string): void
  + getStatus(): TripRequestStatus
  + setStatus(status: TripRequestStatus): void
}

enum TripRequestStatus {
  DRAFT
  PENDING_CONFIRMATION
  CONFIRMED
  SAVED
}

class FieldValidator {
  + validate(tripRequest: TripRequest): vector<string>
  - isEmpty(value: string): bool
  - isValidDateRange(startDate: string, endDate: string): bool
}

class TripRequestRepository {
  - savedRequests: unordered_map<string, TripRequest>
  + save(tripRequest: TripRequest): TripRequest
  + findById(requestId: string): TripRequest
  + findAll(): vector<TripRequest>
}

class TripIntakeController {
  - fieldValidator: FieldValidator
  - tripRequestRepository: TripRequestRepository
  + createTripRequest(requestBody: json): json
  + confirmTripRequest(requestBody: json): json
}

class TripIntakeForm <<frontend>> {
  + submitForm(): void
}

class TripSummary <<frontend>> {
  + displaySummary(tripRequest): void
  + confirmTrip(): void
}
